// Checkout pricing for the shop.
// Amounts are in Japanese yen.

export type LineItem = {
  name?: string;
  price: number; // unit price in yen
  qty: number;
};

export type CheckoutOptions = {
  promoCode?: string;
  isVip?: boolean;
};

export function subtotal(items: LineItem[]): number {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function calculateTotal(items: LineItem[], options: CheckoutOptions = {}): number {
  const sub = subtotal(items);
  const vipDiscountRate = options.isVip ? 0.1 : 0;
  const promoDiscountRate =
    options.promoCode?.trim().toUpperCase() === "BUILD20" && sub >= 3000 ? 0.2 : 0;

  // Use the best eligible discount without stacking, preserving the VIP benefit.
  const discountRate = Math.max(vipDiscountRate, promoDiscountRate);
  return Math.floor(sub * (1 - discountRate));
}
